cmake_minimum_required(VERSION 3.20)
project(axi_StorageStacked VERSION 1.0.0 LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-UNDEBUG)
option(STORAGE_BUILD_CLI "Build the standalone AXI input/output runner" ON)
set(STORAGE_MEMSIM_LIBRARY "" CACHE FILEPATH "Optional externally built mem_sim library")

set(SYSTEMC_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/third_party/systemc)
file(GLOB_RECURSE SYSTEMC_SOURCES CONFIGURE_DEPENDS ${SYSTEMC_ROOT}/src/sysc/*.cpp)
list(FILTER SYSTEMC_SOURCES EXCLUDE REGEX "sc_cor_(pthread|fiber)\\.cpp$")
add_library(storage_systemc STATIC ${SYSTEMC_SOURCES} ${SYSTEMC_ROOT}/src/sysc/qt/qt.c ${SYSTEMC_ROOT}/src/sysc/qt/md/iX86_64.s)
target_include_directories(storage_systemc PUBLIC ${SYSTEMC_ROOT}/src PRIVATE ${SYSTEMC_ROOT}/src/sysc/qt)
target_compile_definitions(storage_systemc PUBLIC SC_INCLUDE_FX SC_INCLUDE_DYNAMIC_PROCESSES)
target_compile_options(storage_systemc PRIVATE -Wno-deprecated-declarations)

if(STORAGE_MEMSIM_LIBRARY)
  add_library(storage_memory SHARED IMPORTED)
  set_target_properties(storage_memory PROPERTIES IMPORTED_LOCATION "${STORAGE_MEMSIM_LIBRARY}")
else()
  enable_testing()
  add_subdirectory(mem_sim)
  add_library(storage_memory ALIAS storagestacked_memsim)
endif()
add_library(storage_axi STATIC storage_axi/aou_backend.cc storage_axi/memsim_backend.cc storage_axi/axi_monitor.cc
  storage_axi/aou_axi2flit.cc storage_axi/aou_flit_packer.cc storage_axi/aou_flit_unpacker.cc)
target_include_directories(storage_axi PUBLIC storage_axi protocol/include axi2flit/systemc/include
  axi2flit/systemc/integration ucie-model/src mem_sim/integration)
target_link_libraries(storage_axi PUBLIC storage_systemc storage_memory pthread)
add_library(StorageStacked::axi ALIAS storage_axi)
if(STORAGE_BUILD_CLI)
  add_executable(axi_storage examples/axi_storage.cc)
  target_link_libraries(axi_storage PRIVATE storage_axi)
endif()

cmake_minimum_required(VERSION 3.20)
project(axi_StorageStacked VERSION 1.0.0 LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-UNDEBUG)
option(STORAGE_BUILD_CLI "Build the standalone AXI input/output runner" ON)
set(STORAGE_MEMSIM_LIBRARY "" CACHE FILEPATH "Optional externally built mem_sim library")

include("${CMAKE_CURRENT_SOURCE_DIR}/../common/cmake/systemc.cmake")

include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/storage_manifest.cmake")

if(STORAGE_MEMSIM_LIBRARY)
  add_library(storage_memory SHARED IMPORTED)
  set_target_properties(storage_memory PROPERTIES IMPORTED_LOCATION "${STORAGE_MEMSIM_LIBRARY}")
else()
  enable_testing()
  add_subdirectory("${STORAGE_MEMSIM_SOURCE}" mem_sim)
  add_library(storage_memory ALIAS storagestacked_memsim)
endif()
add_library(storage_axi STATIC ${STORAGE_SOURCES})
target_include_directories(storage_axi PUBLIC ${STORAGE_INCLUDE_DIRS})
target_compile_features(storage_axi PUBLIC cxx_std_${STORAGE_CXX_STANDARD})
target_compile_definitions(storage_axi PUBLIC SC_INCLUDE_FX SC_INCLUDE_DYNAMIC_PROCESSES)
target_link_libraries(storage_axi PUBLIC storage_systemc storage_memory pthread)
add_library(StorageStacked::axi ALIAS storage_axi)
if(STORAGE_BUILD_CLI)
  add_executable(axi_storage examples/axi_storage.cc)
  find_path(STORAGE_BOOST_INCLUDE boost/property_tree/json_parser.hpp HINTS "$ENV{SS_TOOLCHAIN_PREFIX}/include" "$ENV{SS_PREFIX}/include" REQUIRED)
  target_include_directories(axi_storage PRIVATE "${STORAGE_BOOST_INCLUDE}")
  target_link_libraries(axi_storage PRIVATE storage_axi)
endif()

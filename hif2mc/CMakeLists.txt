cmake_minimum_required(VERSION 3.20)
project(hif2mc VERSION 0.1.0 LANGUAGES CXX)
if(POLICY CMP0144)
  cmake_policy(SET CMP0144 NEW)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(HIF2MC_SANITIZE "Enable address/undefined sanitizers" OFF)
option(HIF2MC_VERILATOR_DEMO "Build the small HIF RTL master demonstration" OFF)
if(HIF2MC_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
set(MEM_SIM_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../mem_sim" CACHE PATH "mem_sim source checkout")
if(NOT EXISTS "${MEM_SIM_DIR}/include/hbm_sim/core/system.hpp")
  message(FATAL_ERROR "MEM_SIM_DIR must name the real mem_sim source checkout")
endif()
add_subdirectory("${MEM_SIM_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/mem_sim" EXCLUDE_FROM_ALL)
add_library(hif2mc src/adapter.cpp src/memsim_backend.cpp)
target_include_directories(hif2mc PUBLIC include)
target_include_directories(hif2mc SYSTEM PUBLIC "${MEM_SIM_DIR}/include")
target_link_libraries(hif2mc PUBLIC hbm_sim_core)
target_compile_options(hif2mc PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_executable(hif2mc_example examples/roundtrip.cpp)
target_link_libraries(hif2mc_example PRIVATE hif2mc)
enable_testing()
add_executable(hif2mc_tests tests/adapter_tests.cpp)
target_link_libraries(hif2mc_tests PRIVATE hif2mc)
target_compile_options(hif2mc_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME hif2mc_contract COMMAND hif2mc_tests contract)
add_test(NAME hif2mc_memsim COMMAND hif2mc_tests memsim)
add_test(NAME hif2mc_roundtrip COMMAND hif2mc_example)
if(HIF2MC_VERILATOR_DEMO)
  find_program(HIF2MC_VERILATOR_EXECUTABLE verilator REQUIRED)
  execute_process(COMMAND "${HIF2MC_VERILATOR_EXECUTABLE}" --getenv VERILATOR_ROOT
                  OUTPUT_VARIABLE HIF2MC_VERILATOR_ROOT OUTPUT_STRIP_TRAILING_WHITESPACE
                  COMMAND_ERROR_IS_FATAL ANY)
  find_package(verilator REQUIRED HINTS "${HIF2MC_VERILATOR_ROOT}" NO_DEFAULT_PATH)
  add_executable(hif2mc_verilator_demo examples/verilator_main.cpp)
  target_link_libraries(hif2mc_verilator_demo PRIVATE hif2mc)
  verilate(hif2mc_verilator_demo SOURCES examples/HifDemoMaster.sv
           TOP_MODULE HifDemoMaster PREFIX VHifDemoMaster TRACE_VCD
           VERILATOR_ARGS --timescale 1ns/1ps)
  add_test(NAME hif2mc_verilator COMMAND hif2mc_verilator_demo)
endif()
